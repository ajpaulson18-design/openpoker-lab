"""Package preserved executed evidence only. No native/math/archive execution."""
from pathlib import Path
from hashlib import sha256
from io import BytesIO
import json
import subprocess
import zipfile

base = Path(__file__).resolve().parent
root = base / 'dyprofile-repo'
destination = base / 'weighted-joint-evidence1'
MAX_MEMBER = MAX_ARCHIVE = 8 << 20
MAX_EXPANDED = 80 << 20


def digest(raw):
    return sha256(raw).hexdigest()


def read(path):
    if not path.is_file() or path.is_symlink() or path.stat().st_size > MAX_MEMBER:
        raise ValueError('bounded regular member: ' + str(path))
    return path.read_bytes()


def tree(directory):
    result = {}
    for path in sorted(directory.rglob('*')):
        if path.is_file():
            name = path.relative_to(directory).as_posix()
            if path.suffix.lower() in ('.exe', '.dll', '.obj', '.pdb', '.etl', '.diagsession') or 'empty-cache' in name:
                raise ValueError('unexpected binary/private/cache evidence member: ' + name)
            result[name] = read(path)
    if not result:
        raise ValueError('empty evidence directory')
    return result


def phase_tree(directory):
    # Every observed complete wire is byte-identical by the frozen benchmark.
    # Keep its bytes once per archive and an authenticated object reference
    # for every original process path. This preserves all132 process receipts
    # without tripling44MB of identical stdout per attempt or raising80MB.
    result = {}
    for path in sorted(directory.rglob('*')):
        if not path.is_file():
            continue
        name = path.relative_to(directory).as_posix()
        raw = read(path)
        if path.name == 'stdout.bin':
            record = json.loads((path.parent / 'process.json').read_text())
            object_name = 'stdout-objects/' + digest(raw) + '.bin'
            if record['streams'][0]['sha256'] != digest(raw) or record['streams'][0]['saved_bytes'] != len(raw):
                raise ValueError('phase stdout object receipt mismatch')
            result[object_name] = raw
            result[name.removesuffix('.bin') + '.ref.json'] = (json.dumps({
                'original_member': name, 'object': object_name, 'bytes': len(raw), 'sha256': digest(raw),
                'encoding': 'Exact original stdout bytes, content addressed deduplication only.'}, indent=2) + '\n').encode()
        else:
            if path.suffix.lower() in ('.etl', '.diagsession', '.exe', '.obj', '.dll', '.pdb'):
                raise ValueError('private/binary phase member')
            result[name] = raw
    return result


def capture_build(name):
    directory = base / name
    return {path.name: read(path) for path in directory.iterdir()
            if path.is_file() and path.name in ('native-exact-build-manifest.json', 'compiler-version.log',
                                               'build.log', 'build-failure.json')}


def logs(*names):
    return {name: read(base / name) for name in names}


def archived_source(commit, names, hashes=None):
    result = {}
    for name in names:
        raw = subprocess.check_output(['git', 'show', commit + ':' + name], cwd=root)
        if hashes and name in hashes:
            variants = (raw, raw.replace(b'\n', b'\r\n')) if b'\r' not in raw else (raw,)
            matches = [value for value in variants if digest(value) == hashes[name]]
            if not matches:
                raise ValueError('recorded raw source not a Git checkout form: ' + name)
            raw = matches[0]
        result['source/' + name] = raw
    return result


def package(name, files):
    files = dict(files)
    if any(not key or '\\' in key or key.startswith('/') or '..' in Path(key).parts for key in files):
        raise ValueError('unsafe archive member')
    inventory = {key: {'bytes': len(raw), 'sha256': digest(raw)} for key, raw in sorted(files.items())}
    files['inner-manifest.json'] = (json.dumps({'schema': 'weighted-joint-evidence-inner-v1', 'files': inventory},
                                            indent=2) + '\n').encode()
    if any(len(raw) > MAX_MEMBER for raw in files.values()):
        raise ValueError('member cap: ' + name)
    memory = BytesIO()
    with zipfile.ZipFile(memory, 'w', compression=zipfile.ZIP_DEFLATED, compresslevel=9) as archive:
        for key, raw in sorted(files.items()):
            info = zipfile.ZipInfo(key, (2026, 11, 24, 0, 0, 0)); info.compress_type = zipfile.ZIP_DEFLATED
            info.create_system = 3; info.external_attr = 0o100644 << 16
            archive.writestr(info, raw)
    raw = memory.getvalue()
    if len(raw) > MAX_ARCHIVE:
        raise ValueError('archive cap: ' + name)
    return raw, {'bytes': len(raw), 'sha256': digest(raw), 'members': len(files),
                 'expanded_bytes': sum(map(len, files.values())), 'inventory': inventory}
